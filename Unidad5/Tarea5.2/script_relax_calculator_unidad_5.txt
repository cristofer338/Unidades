// ============================================================================
// CATÁLOGO DE 10 CONSULTAS FORMALES
// ============================================================================

// C1: Vehículos con kilometraje mayor a 80,000 km
pi vin, placa, marca, modelo_anio (sigma kilometraje > 80000 (vehiculo))

// C2: Refacciones con precio unitario <= 500.00
pi codigo_refaccion, nombre_pieza, precio_unitario (sigma precio_unitario <= 500.00 (refaccion))

// C3: Órdenes con su vehículo y cliente
pi folio_orden, estado, placa, nombres, apellido_paterno, apellido_materno (
  (orden_servicio join orden_servicio.vin = vehiculo.vin vehiculo)
  join vehiculo.id_cliente = cliente.id_cliente cliente
)

// C4: Servicios donde costo_aplicado difiere de precio_base
pi folio_orden, id_servicio, nombre_servicio, costo_aplicado (
  sigma costo_aplicado != precio_base (
    servicio_aplicado join servicio_aplicado.id_servicio = servicio.id_servicio servicio
  )
)

// C5: Datos completos de los mecánicos (especialización)
pi id_empleado, curp, nombres, especialidad, tarifa_hora (
  empleado join empleado.id_empleado = mecanico.id_empleado mecanico
)

// C6: Total de horas trabajadas y órdenes asignadas por mecánico
gamma id_mecanico; sum(horas_trabajadas)->total_horas, count(folio_orden)->num_ordenes (asignacion_trabajo)

// C7: Órdenes cuyo monto total de servicios supera $1,000.00
sigma total_servicios > 1000.00 (
  gamma folio_orden; sum(costo_aplicado)->total_servicios (servicio_aplicado)
)

// C8: Todos los clientes y sus vehículos (incluyendo clientes sin vehículo)
pi id_cliente, nombres, apellido_paterno, vin (
  cliente left outer join cliente.id_cliente = vehiculo.id_cliente vehiculo
)

// C9: Refacciones del catálogo que nunca se han utilizado
pi codigo_refaccion, nombre_pieza (
  refaccion natural join (
    (pi codigo_refaccion (refaccion)) - (pi codigo_refaccion (uso_refaccion))
  )
)

// C10: Importe de mano de obra por mecánico y orden de servicio
pi folio_orden, id_mecanico, nombres, horas_trabajadas * tarifa_hora -> importe_mano_obra (
  (asignacion_trabajo join asignacion_trabajo.id_mecanico = mecanico.id_empleado mecanico)
  join mecanico.id_empleado = empleado.id_empleado empleado
)
